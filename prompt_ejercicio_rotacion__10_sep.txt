¿por qué motivo una llamada a un servidor NGINX que lo único que hace es inyectar un encabezado de tipo x-api-key dejaría al cliente "colgado" o "esperando indefinidamente"?



¿hace falta añadir alguna instrucción especial a mi endpoint de backend o al nginx como tal?



abajo está mi código de NGINX y sus correspondiente archivos en docker

```

# configuración de NGINX

server {

    listen 8090;

    server_name _;



    root /usr/share/nginx/html;

    index index.html;



    # Static frontend — no API key anywhere in these files.

    location / {

        try_files $uri $uri/ =404;

    }



    # The browser sends no credentials; nginx attaches the secret server-side.

    location /api/customers/ {

        # Preserve /api/ when forwarding to the API.

        proxy_pass ${API_UPSTREAM};



        proxy_set_header X-API-Key "${API_KEY}";

        proxy_set_header Host $host;

        proxy_set_header X-Real-IP $remote_addr;

        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;

        proxy_set_header X-Forwarded-Proto $scheme;

    }

}



# códigos de Dockerfile y del entrypoint

FROM nginx:1.27-alpine



# gettext provides envsubst, used to inject API_KEY into the nginx config

# at container start without baking it into the image.

RUN apk add --no-cache gettext



COPY index.html app.js /usr/share/nginx/html/

COPY nginx.conf.template /etc/nginx/templates/nginx.conf.template

COPY docker-entrypoint.sh /docker-entrypoint.sh



RUN chmod +x /docker-entrypoint.sh



EXPOSE 8090

ENTRYPOINT ["/docker-entrypoint.sh"]



#!/bin/sh

set -eu



: "${API_KEY:?API_KEY environment variable must be set (e.g. via docker-compose --env-file .env)}"

: "${API_UPSTREAM:?API_UPSTREAM environment variable must be set}"



# Only substitute the variables we control so nginx's own variables remain intact.

envsubst '${API_KEY} ${API_UPSTREAM}' \

  < /etc/nginx/templates/nginx.conf.template \

  > /etc/nginx/conf.d/default.conf



exec nginx -g 'daemon off;'

```

from fastapi import Request

from fastapi.responses import JSONResponse

from app import create_app

from app.config import API_KEY



app = create_app()



@app.middleware("http")

async def log_request(request: Request, call_next):

    x_api_key = request.headers.get("x-api-key")



    if not x_api_key or x_api_key != API_KEY:

        return JSONResponse(

            status_code=401,

            content={"detail": "Invalid or missing API Key"}

        )



    headers = "; ".join(

        f"{key}={value}"

        for key, value in request.headers.items()

    )



    body = await request.body()



    print(f"HEADERS: {headers}")

    print(f"BODY: {body.decode('utf-8', errors='replace')}")



    response = await call_next(request)



    return response
